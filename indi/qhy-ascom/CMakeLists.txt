cmake_minimum_required(VERSION 3.16)
project(indi_qhy_ascom_style LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(INDI_QHY_VERSION_MAJOR 2)
set(INDI_QHY_VERSION_MINOR 9)
set(INDI_DATA_DIR /usr/share/indi)
configure_file(config.h.cmake config.h)
find_package(Threads REQUIRED)
find_library(INDI_DRIVER indidriver REQUIRED)
find_library(CFITSIO cfitsio REQUIRED)
find_library(NOVA nova REQUIRED)
find_library(USB1 usb-1.0 REQUIRED)
set(QHY_ORIGINAL_SDK /usr/lib/aarch64-linux-gnu/libqhyccd.so.26.2.1 CACHE FILEPATH "Preserved original SDK")
if(NOT EXISTS "${QHY_ORIGINAL_SDK}")
  message(FATAL_ERROR "Original SDK is missing")
endif()
add_library(qhy_driver_objects OBJECT qhy_ccd.cpp qhy_ccd_hotplug_handler.cpp)
target_compile_definitions(qhy_driver_objects PUBLIC CALLBACK_MODE_SUPPORT __CPP_MODE__)
target_include_directories(qhy_driver_objects PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" /usr/include/libindi /usr/include/libqhy /usr/include/libusb-1.0)
add_executable(indi_qhy_ccd $<TARGET_OBJECTS:qhy_driver_objects>)
target_link_options(indi_qhy_ccd PRIVATE -Wl,--no-as-needed)
target_link_libraries(indi_qhy_ccd PRIVATE ${INDI_DRIVER} ${CFITSIO} "${QHY_ORIGINAL_SDK}" ${USB1} ${NOVA} Threads::Threads rt)
add_executable(capture_contract_test capture_contract_test.cpp $<TARGET_OBJECTS:qhy_driver_objects>)
target_compile_definitions(capture_contract_test PRIVATE CALLBACK_MODE_SUPPORT __CPP_MODE__)
target_include_directories(capture_contract_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" /usr/include/libindi /usr/include/libqhy /usr/include/libusb-1.0)
target_link_options(capture_contract_test PRIVATE -Wl,--no-as-needed)
target_link_libraries(capture_contract_test PRIVATE ${INDI_DRIVER} ${CFITSIO} "${QHY_ORIGINAL_SDK}" ${USB1} ${NOVA} Threads::Threads rt)
enable_testing()
add_test(NAME capture_contract COMMAND capture_contract_test)
